pub struct NurbsBasis {
    order: usize,              // Degree + 1
    topology: NurbsTopology,   // Open/Periodic/Closed
    knot_vector: Vec<Real>,
    t_min, t_max: Real,
    spans: Vec<Real>,
    valid: bool,
}

pub struct NurbsCurve {
    name: String,
    basis: NurbsBasis,
    control_points: Vec<Vec4>,  // Homogeneous coordinates
}

pub struct NurbsSurface {
    basis_u, basis_v: NurbsBasis,
    num_control_points_u, num_control_points_v: usize,
    control_points: Vec<Vec4>,
    flip_normals: bool,
}

pub struct CurvePoint {
    valid: bool,
    position: Vec3,
    derivative: Vec3,
}

pub struct SurfacePoint {
    valid: bool,
    position: Vec3,
    derivative_u: Vec3,
    derivative_v: Vec3,
}

use ufbx::nurbs::{NurbsBasis, NurbsCurve, NurbsEvaluator, NurbsTopology};
use ufbx::types::Vec4;

// Create a quadratic Bezier arc
let curve = NurbsCurve {
    name: "arc".to_string(),
    basis: NurbsBasis {
        order: 3,
        topology: NurbsTopology::Open,
        knot_vector: vec![0.0, 0.0, 0.0, 1.0, 1.0, 1.0],
        t_min: 0.0,
        t_max: 1.0,
        spans: vec![0.0, 1.0],
        is_2d: false,
        num_wrap_control_points: 0,
        valid: true,
    },
    control_points: vec![
        Vec4::new(0.0, 0.0, 0.0, 1.0),
        Vec4::new(0.5, 1.0, 0.0, 1.0),
        Vec4::new(1.0, 0.0, 0.0, 1.0),
    ],
};

// Evaluate at t=0.5
let point = NurbsEvaluator::evaluate_curve(&curve, 0.5);
println!("Position: {:?}", point.position);

// Tessellate to 8 segments per span
let points = NurbsEvaluator::tessellate_curve(&curve, 8)?;
